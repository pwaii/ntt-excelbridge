```python
"""Shared tool-adapter helpers: the prompt render + hash every run_toolN duplicated."""
import hashlib
import json

from app.ai.prompts import get_prompt


def render_prompt(tool: str, version: str, fields: tuple[str, ...], inputs: dict) -> str:
    text = get_prompt(tool, version)
    for key in fields:
        value = inputs.get(key, "")
        if not isinstance(value, str):
            value = json.dumps(value)  # serialize lists/dicts; keep caller flexible
        text = text.replace("{" + key + "}", value)
    return text


def sha256_hex(s: str) -> str:
    return hashlib.sha256(s.encode("utf-8")).hexdigest()
```

```python
from app.ai.bedrock_client import BedrockClient
from app.ai.policy import MAX_OUTPUT_TOKENS
from app.domain.schemas import AuditMeta, Tool1Output, ToolResult

# Placeholders the v1 prompt template declares (§10.2 inputs).

    "comment_name", "output_header", "current_headers", "normalized_headers",

def run_tool1(
    inputs: dict, client: BedrockClient | None = None, model_id: str | None = None
```

```python
"""Tool #2 adapter (§10.3, ADR D4): render the type-normalization prompt, call Bedrock,
validate the response into Tool2Output (shape), reject an off-allowlist target type, and
demote over-confident 'safe' rules. Tool #2 returns ONE RULE PER VALUE-PATTERN; the backend
applies each rule to every matching cell and re-parses in Python before acceptance."""
import json
import logging

from app.ai.adapter import render_prompt, sha256_hex
from app.ai.bedrock_client import BedrockClient
from app.ai.policy import (
    CONFIDENCE_FLOOR,
    MAX_OUTPUT_TOKENS,
    estimate_cost_usd,
    safe_log,
)
from app.domain.schemas import AuditMeta, Tool2Output, ToolResult
from app.settings import settings

TOOL = "tool2_types"
PROMPT_VERSION = "v1"

# Placeholders the v1 prompt template declares (§10.3 inputs).
_FIELDS = ("column_id", "target_type", "samples", "type_distribution", "constraints")

# The vocabulary the rule engine must use when it proposes a target type (handoff to BE).
ALLOWED_TARGET_TYPES = frozenset(
    {"string", "integer", "number", "date", "datetime", "boolean"}
)

_log = logging.getLogger(__name__)


def _validate_against_inputs(payload: Tool2Output) -> None:
    """Reject the whole response (invariant #1) if the model used an unknown target type.
    Rules describe value-patterns, not prompt rows, so there is no row-id set to check —
    the backend re-parses every applied cell, which is the real safety gate (§10.3)."""
    if payload.target_type not in ALLOWED_TARGET_TYPES:
        raise ValueError(
            f"target_type {payload.target_type!r} not in {set(ALLOWED_TARGET_TYPES)}"
        )


def _apply_confidence_floor(payload: Tool2Output) -> None:
    """Demote-only guardrail: a 'safe' rule below the floor becomes 'ambiguous' (manual
    review). confidence/status are self-reported, so the guard only adds caution."""
    for r in payload.rules:
        if r.status == "safe" and r.confidence < CONFIDENCE_FLOOR:
            r.status = "ambiguous"


def run_tool2(
    inputs: dict, client: BedrockClient | None = None, model_id: str | None = None
) -> ToolResult[Tool2Output]:
    client = client or BedrockClient()
    model_id = model_id or settings.bedrock_model_id

    prompt = render_prompt(TOOL, PROMPT_VERSION, _FIELDS, inputs)
    data, raw, usage, latency_ms = client.converse_json(
        model_id, prompt, MAX_OUTPUT_TOKENS, settings.bedrock_temperature
    )
    payload = Tool2Output(**data)  # trust boundary: reject anything off-shape
    _validate_against_inputs(payload)  # reject off-allowlist target type
    _apply_confidence_floor(payload)   # demote-only guardrail

    meta = AuditMeta(
        tool=TOOL, model_id=model_id, prompt_version=PROMPT_VERSION,
        input_hash=sha256_hex(prompt), output_hash=sha256_hex(raw), latency_ms=latency_ms,
        input_tokens=usage["inputTokens"], output_tokens=usage["outputTokens"],
    )
    _log.info(json.dumps(safe_log({
        "tool": TOOL, "model_id": model_id, "prompt_version": PROMPT_VERSION,
        "input_hash": meta.input_hash, "output_hash": meta.output_hash,
        "latency_ms": latency_ms, "input_tokens": meta.input_tokens,
        "output_tokens": meta.output_tokens, "rule_count": len(payload.rules),
        "estimated_usd": estimate_cost_usd(meta.input_tokens, meta.output_tokens),
    })))
    return ToolResult[Tool2Output](meta=meta, payload=payload)
```

```python
Tool3Output, then reject-whole on a hallucinated referenced header, a non-'=' formula, or a
forbidden function. The authoritative function allowlist + in-range check stay on the backend;
this is a thin caution guard. Labeled 'syntax verified,' never 'calculated.'"""
import json
import logging

from app.ai.adapter import render_prompt, sha256_hex
from app.ai.bedrock_client import BedrockClient
from app.ai.policy import MAX_OUTPUT_TOKENS, estimate_cost_usd, safe_log
from app.domain.schemas import AuditMeta, Tool3Output, ToolResult
from app.settings import settings

TOOL = "tool3_formulas"
PROMPT_VERSION = "v1"

# Placeholders the v1 prompt template declares (§10.4 inputs).
_FIELDS = ("meaning", "source_headers", "allowed_functions", "samples", "expected_type")

# Thin blocklist mirroring the §10.4 bans; BE owns the authoritative allowlist + range check.
# Matched as "TOKEN(" so a header like "Snowfall" never trips the "NOW" substring.
FORBIDDEN_TOKENS = frozenset(
    {"INDIRECT", "OFFSET", "WEBSERVICE", "HYPERLINK", "NOW", "TODAY", "RAND", "RANDBETWEEN"}
)

_log = logging.getLogger(__name__)


def _validate_against_inputs(payload: Tool3Output, inputs: dict) -> None:
    """Reject the whole response (invariant #1) on a value that isn't a formula, references a
    header never sent, or calls a forbidden function."""
    if not payload.formula_template.startswith("="):
        raise ValueError("formula_template must start with '='")
    allowed = set(inputs.get("source_headers", []))
    unknown = [h for h in payload.referenced_headers if h not in allowed]
    if unknown:
        raise ValueError(f"referenced_headers not in source_headers: {unknown}")
    upper = payload.formula_template.upper()
    hits = [t for t in FORBIDDEN_TOKENS if f"{t}(" in upper]
    if hits:
        raise ValueError(f"forbidden functions in formula: {hits}")


def run_tool3(
    inputs: dict, client: BedrockClient | None = None, model_id: str | None = None
) -> ToolResult[Tool3Output]:
    client = client or BedrockClient()
    model_id = model_id or settings.bedrock_model_id_heavy

    prompt = render_prompt(TOOL, PROMPT_VERSION, _FIELDS, inputs)
    data, raw, usage, latency_ms = client.converse_json(
        model_id, prompt, MAX_OUTPUT_TOKENS, settings.bedrock_temperature
    )
    payload = Tool3Output(**data)  # trust boundary: reject anything off-shape
    _validate_against_inputs(payload, inputs)  # reject hallucinated header / forbidden formula

    meta = AuditMeta(
        tool=TOOL, model_id=model_id, prompt_version=PROMPT_VERSION,
        input_hash=sha256_hex(prompt), output_hash=sha256_hex(raw), latency_ms=latency_ms,
        input_tokens=usage["inputTokens"], output_tokens=usage["outputTokens"],
    )
    _log.info(json.dumps(safe_log({
        "tool": TOOL, "model_id": model_id, "prompt_version": PROMPT_VERSION,
        "input_hash": meta.input_hash, "output_hash": meta.output_hash,
        "latency_ms": latency_ms, "input_tokens": meta.input_tokens,
        "output_tokens": meta.output_tokens,
        "estimated_usd": estimate_cost_usd(meta.input_tokens, meta.output_tokens),
    })))
    return ToolResult[Tool3Output](meta=meta, payload=payload)
```

```python
"""Tool #4 adapter (§10.5): render the standardization prompt, call Bedrock, validate into
Tool4Output, reject-whole on a canonical value the prompt never offered, then demote one-to-many
and over-confident 'safe' mappings to 'ambiguous'. The model may only map to supplied candidates."""
import json
import logging

from app.ai.adapter import render_prompt, sha256_hex
from app.ai.bedrock_client import BedrockClient
from app.ai.policy import (
    CONFIDENCE_FLOOR,
    MAX_OUTPUT_TOKENS,
    estimate_cost_usd,
    safe_log,
)
from app.domain.schemas import AuditMeta, Tool4Output, ToolResult
from app.settings import settings

TOOL = "tool4_standardize"
PROMPT_VERSION = "v1"

# Placeholders the v1 prompt template declares (§10.5 inputs).
_FIELDS = ("source_values", "canonical_candidates", "samples")

_log = logging.getLogger(__name__)


def _validate_against_inputs(payload: Tool4Output, inputs: dict) -> None:
    """Reject the whole response (invariant #1) if the model invented a canonical value. An
    unmapped mapping must carry an empty canonical_value; anything else must be a supplied
    candidate."""
    allowed = set(inputs.get("canonical_candidates", []))
    for m in payload.mappings:
        if m.cardinality == "unmapped" and m.canonical_value == "":
            continue
        if m.canonical_value not in allowed:
            raise ValueError(f"canonical_value {m.canonical_value!r} not in candidates")


def _apply_guardrails(payload: Tool4Output) -> None:
    """Demote-only: a source_value mapped to >1 distinct canonical_value is one-to-many (always
    ambiguous); a 'safe' below the floor is demoted too."""
    by_source: dict[str, set[str]] = {}
    for m in payload.mappings:
        by_source.setdefault(m.source_value, set()).add(m.canonical_value)
    for m in payload.mappings:
        if len(by_source[m.source_value]) > 1:
            m.action = "ambiguous"
        elif m.action == "safe" and m.confidence < CONFIDENCE_FLOOR:
            m.action = "ambiguous"


def run_tool4(
    inputs: dict, client: BedrockClient | None = None, model_id: str | None = None
) -> ToolResult[Tool4Output]:
    client = client or BedrockClient()
    model_id = model_id or settings.bedrock_model_id_heavy

    prompt = render_prompt(TOOL, PROMPT_VERSION, _FIELDS, inputs)
    data, raw, usage, latency_ms = client.converse_json(
        model_id, prompt, MAX_OUTPUT_TOKENS, settings.bedrock_temperature
    )
    payload = Tool4Output(**data)  # trust boundary: reject anything off-shape
    _validate_against_inputs(payload, inputs)  # reject invented canonical values
    _apply_guardrails(payload)                 # demote one-to-many / low-confidence

    meta = AuditMeta(
        tool=TOOL, model_id=model_id, prompt_version=PROMPT_VERSION,
        input_hash=sha256_hex(prompt), output_hash=sha256_hex(raw), latency_ms=latency_ms,
        input_tokens=usage["inputTokens"], output_tokens=usage["outputTokens"],
    )
    _log.info(json.dumps(safe_log({
        "tool": TOOL, "model_id": model_id, "prompt_version": PROMPT_VERSION,
        "input_hash": meta.input_hash, "output_hash": meta.output_hash,
        "latency_ms": latency_ms, "input_tokens": meta.input_tokens,
        "output_tokens": meta.output_tokens, "rule_count": len(payload.mappings),
        "estimated_usd": estimate_cost_usd(meta.input_tokens, meta.output_tokens),
    })))
    return ToolResult[Tool4Output](meta=meta, payload=payload)
```
