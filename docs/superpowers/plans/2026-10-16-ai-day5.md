```python
return 'unmatched'."""

import pytest
from pydantic import ValidationError

from app.ai.tool5_update import run_tool5
from app.domain.schemas import Tool5Output, ToolResult


class FakeBedrock:
    """Stands in for a boto3 bedrock-runtime client. Records the converse() kwargs."""

    def __init__(self, text: str, usage: dict | None = None):
        self._text = text
        self._usage = usage or {"inputTokens": 10, "outputTokens": 5}
        self.calls: list[dict] = []

    def converse(self, **kwargs):
        self.calls.append(kwargs)
        return {
            "output": {"message": {"content": [{"text": self._text}]}},
            "usage": self._usage,
        }


def _client(text: str, usage: dict | None = None):
    from app.ai.bedrock_client import BedrockClient

    return BedrockClient(client=FakeBedrock(text, usage))


T5_FIXTURE = {
    "current_rows": [
        {"current_row_id": "c1", "fields": {"Email": "a@x.com"}},
        {"current_row_id": "c2", "fields": {"Email": "zzz@x.com"}},
    ],
    "candidate_rows": [{"main_row_id": "m1", "fields": {"Email": "a@x.com"}}],
    "match_key": "Email",
}

# One confident exact match + one confident no-match (neither gets demoted).
GOOD_T5 = (
    '{"matches": ['

def test_run_tool5_valid_match_wraps_meta():

    assert isinstance(result, ToolResult)
    assert isinstance(result.payload, Tool5Output)
    assert result.payload.matches[0].selected_main_row_id == "m1"

    assert result.meta.prompt_version == "v1"
    assert len(result.meta.input_hash) == 64 and len(result.meta.output_hash) == 64
    assert result.meta.input_tokens == 10 and result.meta.output_tokens == 5

    run_tool5(T5_FIXTURE, client=client)
    text = client._client.calls[0]["messages"][0]["content"][0]["text"]
    assert "{current_rows}" not in text  # fully rendered
    assert "m1" in text                  # candidate id serialized into the prompt

        '{"matches": [{"current_row_id": "c1", "candidate_main_row_ids": ["m1"], '
```

```python
"""Tool #5 adapter (§10.6): render the unresolved-match prompt, call Bedrock, validate into
Tool5Output, reject-whole on a selected row ID that was never offered (or an 'unmatched' row
that still carries a selection), then demote ambiguous/duplicate/conflict/low-confidence rows
to 'human_review'. The model may only SELECT from the candidate IDs supplied per row — it never
invents one. BE re-rejects any off-candidate ID at apply-time and owns conflict resolution."""
import json
import logging

from app.ai.adapter import render_prompt, sha256_hex
from app.ai.bedrock_client import BedrockClient
from app.ai.policy import (
    CONFIDENCE_FLOOR,
    MAX_OUTPUT_TOKENS,
    estimate_cost_usd,
    safe_log,
)
from app.domain.schemas import AuditMeta, Tool5Output, ToolResult
from app.settings import settings

TOOL = "tool5_update"
PROMPT_VERSION = "v1"

# Placeholders the v1 prompt template declares (§10.6 inputs).
_FIELDS = ("current_rows", "candidate_rows", "match_key")

# match_types that are never auto-applied regardless of confidence (prompt + §10.6).
_HUMAN_ONLY_TYPES = frozenset({"ambiguous", "duplicate", "conflict"})

_log = logging.getLogger(__name__)


def _validate_selections(payload: Tool5Output) -> None:
    """Reject the whole response (invariant #1) if a match selects a row ID outside the
    candidates it was offered, or marks a row 'unmatched' while still selecting one."""
    for m in payload.matches:
        if m.selected_main_row_id is not None and (
            m.selected_main_row_id not in m.candidate_main_row_ids
        ):
            raise ValueError(
                f"selected_main_row_id {m.selected_main_row_id!r} not in candidate list"
            )
        if m.match_type == "unmatched" and m.selected_main_row_id is not None:
            raise ValueError("unmatched match must have null selected_main_row_id")


def _apply_guardrails(payload: Tool5Output) -> None:
    """Demote-only: send ambiguous/duplicate/conflict and below-floor matches to human review.
    Never clears an action the model already set to review."""
    for m in payload.matches:
        if m.match_type in _HUMAN_ONLY_TYPES or m.confidence < CONFIDENCE_FLOOR:
            m.recommended_action = "human_review"


def run_tool5(
    inputs: dict, client: BedrockClient | None = None, model_id: str | None = None
) -> ToolResult[Tool5Output]:
    client = client or BedrockClient()
    model_id = model_id or settings.bedrock_model_id  # Nova Lite: structured selection, not heavy reasoning

    prompt = render_prompt(TOOL, PROMPT_VERSION, _FIELDS, inputs)
    data, raw, usage, latency_ms = client.converse_json(
        model_id, prompt, MAX_OUTPUT_TOKENS, settings.bedrock_temperature
    )
    payload = Tool5Output(**data)  # trust boundary: reject anything off-shape
    _validate_selections(payload)  # reject un-offered IDs / contradictory unmatched
    _apply_guardrails(payload)     # demote to human_review

    meta = AuditMeta(
        tool=TOOL, model_id=model_id, prompt_version=PROMPT_VERSION,
        input_hash=sha256_hex(prompt), output_hash=sha256_hex(raw), latency_ms=latency_ms,
        input_tokens=usage["inputTokens"], output_tokens=usage["outputTokens"],
    )
    _log.info(json.dumps(safe_log({
        "tool": TOOL, "model_id": model_id, "prompt_version": PROMPT_VERSION,
        "input_hash": meta.input_hash, "output_hash": meta.output_hash,
        "latency_ms": latency_ms, "input_tokens": meta.input_tokens,
        "output_tokens": meta.output_tokens, "rule_count": len(payload.matches),
        "estimated_usd": estimate_cost_usd(meta.input_tokens, meta.output_tokens),
    })))
    return ToolResult[Tool5Output](meta=meta, payload=payload)
```

## Notes / handoff

- **BE handoff:** `run_tool5` guards are *caution only*. BE still (§10.6) re-rejects any `selected_main_row_id` not in the candidate list at apply-time, owns candidate generation (ADR D6 key-blocking + fuzzy, top ~5, recall-biased — plain Python, no new library), **skips** unmatched current rows for the demo (ADR D7; insert is future work, `memory/future-work-insert-unmatched-rows.md`), and owns the duplicate/conflict resolution policy (merge / latest-wins / keep-both / reject-all). The AI only labels + recommends.
- **The candidate check reads each match's own `candidate_main_row_ids`.** The model echoes back the IDs it was offered; this is a self-consistency guard, deliberately thinner than BE's authoritative re-check against the candidate set it generated. That is intentional per the spec — the real gate is on the backend.
- **Model:** Nova Lite (`settings.bedrock_model_id`), matching Tools #1/#2 — selecting among ~5 bounded candidates is not heavy-reasoning work like formula generation (#3/#4).
- **Threshold tuning** ("ambiguous → humans") is wired to the shared `CONFIDENCE_FLOOR = 0.80`; tuning it against *real* confidences is Day 6, not this plan.
- **This completes Tools #1–#5.** Day 6 is the real-Bedrock fixture evaluation + token/latency measurement + response caching by input hash. End-to-end product wiring (a worker calling `run_toolN`, feeding the review gates) is a separate BE integration task, not in the AI day plan.
